def print_calendar(n_days, start_weekday):
    # Словарь для преобразования названий дней в индексы
    weekdays = {
        'Monday': 0,
        'Tuesday': 1,
        'Wednesday': 2,
        'Thursday': 3,
        'Friday': 4,
        'Saturday': 5,
        'Sunday': 6
    }

    # Получаем индекс первого дня месяца
    start_index = weekdays[start_weekday]

    # Создаем список для хранения строк с днями
    weeks = []
    current_week = ['..'] * 7  # Инициализируем первую неделю пустыми значениями

    # Заполняем первую неделю
    for day in range(1, n_days + 1):
        if day == 1:
            current_week[start_index] = f".."
        current_week[(start_index + day - 1) % 7] = f".{day}" if day < 10 else str(day)

        # Если неделя заполнена, добавляем её в список и начинаем новую
        if (start_index + day) % 7 == 0:
            weeks.append(current_week)
            current_week = ['..'] * 7  # Новая неделя

    # Добавляем последнюю неделю, если она не пустая
    if any(day != '..' for day in current_week):
        weeks.append(current_week)

    # Печатаем результат
    for week in weeks:
        print(" ".join(week))

# Пример использования
input_data = input().strip().split()
n_days = int(input_data[0])
start_weekday = input_data[1]

print_calendar(n_days, start_weekday)
